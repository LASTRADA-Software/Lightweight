93/// Binds a single @c char as a one-character @c CHAR(1) value.
94///
95/// Unlike the numeric primitives, @c char maps to a character type (@c SQL_C_CHAR / @c SQL_CHAR), whose
96/// ODBC contract differs from fixed-width data in three ways, each of which this binder honours:
97/// - On input, the column size is the length in characters and must not be zero (MS SQL Server rejects
98/// zero with HY104 "Invalid precision value").
99/// - On input, the value is not NUL-terminated, so its length must be passed through the indicator —
100/// without one the driver reads the value as a NUL-terminated string, past the end of the @c char.
101/// - On output, the driver NUL-terminates what it writes, so the buffer needs room for one more byte
102/// than the value; a one-byte buffer receives nothing.
103template <>
104struct SqlDataBinder<char>
105{
106 /// The column type a @c char maps to.
107 static constexpr SqlColumnTypeDefinition ColumnType = SqlColumnTypeDefinitions::Char { 1 };
108
109 /// Binds a single @c char as an input parameter.
110 ///
111 /// @param stmt The ODBC statement handle.
112 /// @param column The 1-based parameter index.
113 /// @param value The character to bind; it must outlive the statement's execution.
114 /// @param cb Provides the length indicator, which must also outlive the execution.
115 /// @return The ODBC return code of @c SQLBindParameter.
116 static LIGHTWEIGHT_FORCE_INLINE SQLRETURN InputParameter(SQLHSTMT stmt,
117 SQLUSMALLINT column,
118 char const& value,
119 SqlDataBinderCallback& cb) noexcept
120 {
121 auto* const indicator = cb.ProvideInputIndicator();
122 *indicator = CharLength;
123 return SQLBindParameter(
124 stmt, column, SQL_PARAM_INPUT, SQL_C_CHAR, SQL_CHAR, ColumnSize, 0, (SQLPOINTER) &value, 0, indicator);
125 }
126
127 /// Binds a contiguous (column-wise) array of characters as an input parameter.
128 ///
129 /// @param stmt The ODBC statement handle.
130 /// @param column The 1-based parameter index.
131 /// @param values The first of @p rowCount characters; they must outlive the statement's execution.
132 /// @param rowCount The number of rows in the batch.
133 /// @param cb Provides the per-row length indicators when @p indicators is not given.
134 /// @param indicators Optional per-row indicator array (the length, or @c SQL_NULL_DATA). When null,
135 /// every row is bound as one character.
136 /// @return The ODBC return code of @c SQLBindParameter.
137 static LIGHTWEIGHT_FORCE_INLINE SQLRETURN BatchInputParameter(SQLHSTMT stmt,
138 SQLUSMALLINT column,
139 char const* values,
140 size_t rowCount,
141 SqlDataBinderCallback& cb,
142 SQLLEN* indicators = nullptr) noexcept
143 {
144 if (!indicators)
145 {
146 indicators = cb.ProvideInputIndicators(rowCount);
147 std::ranges::fill(std::span { indicators, rowCount }, CharLength);
148 }
149 return SQLBindParameter(stmt,
150 column,
151 SQL_PARAM_INPUT,
152 SQL_C_CHAR,
153 SQL_CHAR,
154 ColumnSize,
155 0,
156 (SQLPOINTER) values,
157 sizeof(char),
158 indicators);
159 }
160
161 /// Binds @p result as the output column @p column, filled in on every fetch.
162 ///
163 /// The driver writes into a NUL-terminated staging buffer, which a post-process callback copies into
164 /// @p result; a NULL leaves @p result untouched (@c std::optional<char> resets itself on it).
165 ///
166 /// @param stmt The ODBC statement handle.
167 /// @param column The 1-based column index.
168 /// @param result Receives the character; it must outlive the fetches.
169 /// @param indicator Receives the length, or @c SQL_NULL_DATA.
170 /// @param cb Holds the staging buffer and runs the copy after each fetch.
171 /// @return The ODBC return code of @c SQLBindCol.
172 static SQLRETURN OutputColumn(
173 SQLHSTMT stmt, SQLUSMALLINT column, char* result, SQLLEN* indicator, SqlDataBinderCallback& cb) noexcept
174 {
175 auto buffer = std::make_shared<Buffer>();
176 auto const sqlReturn = SQLBindCol(stmt, column, SQL_C_CHAR, buffer->data(), BufferSize, indicator);
177 cb.PlanPostProcessOutputColumn([buffer, result, indicator]() {
178 if (*indicator != SQL_NULL_DATA)
179 *result = buffer->front();
180 });
181 return sqlReturn;
182 }
183
184 /// Reads the character in column @p column of the current row.
185 ///
186 /// @param stmt The ODBC statement handle.
187 /// @param column The 1-based column index.
188 /// @param result Receives the character; left untouched on NULL.
189 /// @param indicator Receives the length, or @c SQL_NULL_DATA.
190 /// @return The ODBC return code of @c SQLGetData.
191 static SQLRETURN GetColumn(
192 SQLHSTMT stmt, SQLUSMALLINT column, char* result, SQLLEN* indicator, SqlDataBinderCallback const& /*cb*/) noexcept
193 {
194 auto buffer = Buffer {};
195 auto const sqlReturn = SQLGetData(stmt, column, SQL_C_CHAR, buffer.data(), BufferSize, indicator);
196 if (SQL_SUCCEEDED(sqlReturn) && *indicator != SQL_NULL_DATA)
197 *result = buffer.front();
198 return sqlReturn;
199 }
200
201 /// @param value The character to render.
202 /// @return @p value as a one-character string.
203 static LIGHTWEIGHT_FORCE_INLINE std::string Inspect(char value)
204 {
205 return std::string(1, value);
206 }
207
208 private:
209 // One character plus the NUL terminator the driver appends to SQL_C_CHAR output.
210 using Buffer = std::array<char, 2>;
211 static constexpr SQLLEN CharLength = 1;
212 static constexpr SQLULEN ColumnSize = 1;
213 static constexpr SQLLEN BufferSize = sizeof(Buffer);
214};

239// These fixed-width primitives bind via a plain SQLBindParameter and are eligible for native row-wise
240// batch binding (see SqlIsNativeRowBindableValue in Core.hpp). A single constrained partial
241// specialization, keyed on detail::IsAnyOf, covers them all.
242//
243// char is deliberately absent: row-wise binding strides the value in place, but a char needs a per-row
244// length on input and a NUL-terminated staging buffer on output (see SqlDataBinder<char>), so records
245// carrying one take the per-row paths instead.
246template <typename T>
247 requires detail::IsAnyOf<T, bool, int8_t, uint8_t, int16_t, uint16_t, int32_t, uint32_t, int64_t, uint64_t,
248 float, double
249#if !defined(_WIN32) && !defined(__APPLE__)
250 , long long, unsigned long long
251#endif
252#if defined(__APPLE__)
253 , std::size_t
254#endif
255 >
256inline constexpr bool SqlIsNativeRowBindableValue<T> = true;
