22/// @brief This API represents a many-to-many relationship between two records through a third record.
23///
24/// The join record is named with the @ref Through marker, so that the reader can tell it apart from
25/// the referenced record at a glance:
26///
27/// @code
28/// struct Friendship;
29/// struct Human
30/// {
31/// Field<int, PrimaryKey::AutoAssign> id;
32/// HasManyThrough<Human, Through<Friendship>> friends;
33/// };
34/// @endcode
35///
36/// The join record must declare one `BelongsTo` pointing back at the record owning this relationship,
37/// and one pointing at the referenced record. Both are located by matching the relationship *type*.
38///
39/// When the join record cannot be resolved that way - most notably a self-referential many-to-many,
40/// where both of its foreign keys point at the *same* table - name the two foreign key columns
41/// explicitly:
42///
43/// @code
44/// struct Friendship;
45/// struct Human
46/// {
47/// Field<int, PrimaryKey::AutoAssign> id;
48/// HasManyThrough<Human, Through<Friendship>, SqlRealName { "a_id" }, SqlRealName { "b_id" }> friends;
49/// };
50/// struct Friendship
51/// {
52/// Field<int, PrimaryKey::AutoAssign> id;
53/// BelongsTo<&Human::id, SqlRealName { "a_id" }> a;
54/// BelongsTo<&Human::id, SqlRealName { "b_id" }> b;
55/// };
56/// @endcode
57///
58/// @tparam ReferencedRecordT The record type on the "many" side of the relationship.
59/// @tparam ThroughSpec The join record, wrapped as `Through<T>`. Naming the record bare is deprecated.
60/// @tparam TheOwnerSelector Singles out the join record's foreign key pointing at the *owning* record.
61/// @tparam TheReferencedSelector Singles out the join record's foreign key pointing at @p ReferencedRecordT.
62///
63/// @see DataMapper, Field, HasMany, Through, RelationSelector
64/// @ingroup DataMapper
65template <typename ReferencedRecordT,
66 typename ThroughSpec,
67 auto TheOwnerSelector = AutoDetectRelation,
68 auto TheReferencedSelector = AutoDetectRelation>
69class HasManyThrough
70{
71 static_assert(RelationSelector<TheOwnerSelector> && RelationSelector<TheReferencedSelector>,
72 "The selector template arguments of HasManyThrough must be foreign key column names "
73 "(a SqlRealName) or std::nullopt to resolve the relationship automatically.");
74
75 static_assert(!IsThrough<ReferencedRecordT>,
76 "The referenced record of HasManyThrough must not be wrapped in Through<>, "
77 "only the join record is.");

102 /// @brief Retrieves the records, loading them on first access.
103 ///
104 /// Never throws for unavailable records: the reason comes back as the error instead.
105 ///
106 /// @return The records; or @ref RelationError::NotConfigured when there is no loader (a hand-built
107 /// record, or one read with `loadRelations = false`), @ref RelationError::Outdated when the
108 /// default connection string changed since the record was read, @ref RelationError::QueryFailed
109 /// when the load query failed.
110 [[nodiscard]] RelationResult<std::reference_wrapper<ReferencedRecordList const>> All() const
111 {
112 return Load().transform([this] { return std::reference_wrapper<ReferencedRecordList const> { *_records }; });
113 }

130 /// @return The number of records in this relationship - counted by a query, without loading them,
131 /// unless they are loaded already - or why it is unavailable (see @ref All()).
132 [[nodiscard]] RelationResult<std::size_t> Count() const
133 {
134 if (_records)
135 return _records->size();
136 if (_count)
137 return *_count;
138 if (_loadError)
139 return std::unexpected { *_loadError };
140 if (!_loader.count)
141 return std::unexpected { RelationError::NotConfigured };
142
143 auto counted = _loader.count();
144 if (counted)
145 _count = *counted;
146 else if (counted.error() != RelationError::QueryFailed)
147 _loadError = counted.error();
148 return counted;
149 }
150
151 /// @return Whether this relationship is empty, or why that is unknown (see @ref All()).
152 [[nodiscard]] RelationResult<bool> IsEmpty() const
153 {
154 return Count().transform([](std::size_t count) { return count == 0; });
155 }

219 /// Carries the deferred loads, installed by the DataMapper.
220 struct Loader
221 {
222 /// Counts the records without loading them.
223 std::function<RelationResult<size_t>()> count;
224 /// Loads all records.
225 std::function<RelationResult<ReferencedRecordList>()> all;
226 /// Streams the records to a callback.
227 std::function<RelationResult<void>(std::function<void(ReferencedRecord const&)>)> each;
228 };
229
230 /// Used internally to configure on-demand loading of the records.
231 void SetAutoLoader(Loader loader) noexcept
232 {
233 _loader = std::move(loader);
234 }
235
236 /// @brief Loads the records from the database again, forgetting what was loaded or known before.
237 /// @return Nothing, or why the records are unavailable (see @ref All()).
238 RelationResult<void> Reload()
239 {
240 _count = std::nullopt;
241 _records = std::nullopt;
242 _loadError.reset();
243 return Load();
244 }
245
246 /// @brief Calls @p callable for each record, without holding them all in memory when not loaded yet.
247 ///
248 /// An exception thrown by @p callable propagates unchanged.
249 ///
250 /// @param callable Called once per record.
251 /// @return Nothing, or why the records are unavailable (see @ref All()).
252 template <typename Callable>
253 RelationResult<void> Each(Callable const& callable)
254 {
255 if (!_records && !_loadError && _loader.each)
256 {
257 auto streamed = _loader.each(callable);
258 if (!streamed && streamed.error() != RelationError::QueryFailed)
259 _loadError = streamed.error();
260 return streamed;
261 }
262
263 return All().transform([&callable](ReferencedRecordList const& records) {
264 for (auto const& record: records)
265 callable(*record);
266 });
267 }
268
269 private:
270 /// Loads the records unless they already are, or their unavailability is already known.
271 ///
272 /// @ref RelationError::Outdated is remembered - it does not change by asking again - until records
273 /// are emplaced or reloaded; a failed query is retried on the next access.
274 [[nodiscard]] RelationResult<void> Load() const
275 {
276 if (_records)
277 return {};
278 if (_loadError)
279 return std::unexpected { *_loadError };
280 if (!_loader.all)
281 return std::unexpected { RelationError::NotConfigured };
282
283 auto loaded = _loader.all();
284 if (!loaded)
285 {
286 if (loaded.error() != RelationError::QueryFailed)
287 _loadError = loaded.error();
288 return std::unexpected { loaded.error() };
289 }
290 _records = std::move(*loaded);
291 return {};
292 }
293
294 /// @return The loaded records, or throws SqlRequireLoadedError.
295 [[nodiscard]] ReferencedRecordList& LoadOrThrow() const
296 {
297 if (auto const loaded = Load(); !loaded)
298 throw SqlRequireLoadedError(Reflection::TypeNameOf<std::remove_cvref_t<decltype(*this)>>, loaded.error());
299 return *_records; // NOLINT(bugprone-unchecked-optional-access)
300 }
301
302 Loader _loader;
303
304 mutable std::optional<size_t> _count;
305 mutable std::optional<ReferencedRecordList> _records;
306 mutable std::optional<RelationError> _loadError;
307};
