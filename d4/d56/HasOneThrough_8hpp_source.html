17/// @brief Represents a one-to-one relationship through a join table.
18///
19/// The `OtherTable` parameter is the record reached through the join table.
20/// The `ThroughTable` parameter is the join table, which references the current record.
21///
22/// Both foreign keys are located by matching the relationship *type*. When either record holds more
23/// than one foreign key into the same table, name the column to single one out - see
24/// the RelationSelector concept and the example on @ref HasManyThrough.
25///
26/// @tparam OtherTable The record type reached through the join table.
27/// @tparam ThroughTable The join record type, holding a foreign key back to the owning record.
28/// @tparam TheOwnerSelector Singles out the join record's foreign key pointing at the *owning* record.
29/// @tparam TheThroughSelector Singles out @p OtherTable's foreign key pointing at @p ThroughTable.
30///
31/// @see DataMapper, Field, HasManyThrough, RelationSelector
32/// @ingroup DataMapper
33template <typename OtherTable,
34 typename ThroughTable,
35 auto TheOwnerSelector = AutoDetectRelation,
36 auto TheThroughSelector = AutoDetectRelation>
37class HasOneThrough
38{
39 static_assert(RelationSelector<TheOwnerSelector> && RelationSelector<TheThroughSelector>,
40 "The selector template arguments of HasOneThrough must be foreign key column names "
41 "(a SqlRealName) or std::nullopt to resolve the relationship automatically.");
42
43 public:
44 /// The record type of the "through" side of the relationship.
45 using ThroughRecord = ThroughTable;
46
47 /// The record type of the "Other" side of the relationship.
48 using ReferencedRecord = OtherTable;
49
50 /// Singles out the join record's foreign key pointing at the record owning this relationship.
51 static constexpr auto OwnerSelector = TheOwnerSelector;
52
53 /// Singles out @ref ReferencedRecord's foreign key pointing at @ref ThroughRecord.
54 static constexpr auto ThroughSelector = TheThroughSelector;
55

91 std::weak_ordering operator<=>(HasOneThrough const& other) const noexcept = default;
92
93 struct Loader
94 {
95 std::function<std::shared_ptr<ReferencedRecord>()> loadReference {};
96 };
97
98 /// Used internally to configure on-demand loading of the record.
99 void SetAutoLoader(Loader loader)
100 {
101 _loader = std::move(loader);
102 }
103
104 private:
105 void RequireLoaded() const
106 {
107 if (IsLoaded())
108 return;
109
110 if (_loader.loadReference)
111 _record = _loader.loadReference();
112
113 if (!IsLoaded())
114 throw SqlRequireLoadedError { Reflection::TypeNameOf<std::remove_cvref_t<decltype(*this)>> };
115 }
116
117 Loader _loader {};
118
119 // We use shared_ptr to not require ReferencedRecord to be declared before HasOneThrough.
120 mutable std::shared_ptr<ReferencedRecord> _record {};
121};
