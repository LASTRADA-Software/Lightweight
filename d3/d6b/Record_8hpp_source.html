112/// @brief Selector value meaning "resolve the relationship automatically; the match must be unique".
113///
114/// This is the default for every relationship selector. Pass a `SqlRealName` instead to name the
115/// foreign key column explicitly, which is required when a record holds more than one foreign key
116/// into the same table.
117///
118/// @ingroup DataMapper
119inline constexpr std::nullopt_t AutoDetectRelation = std::nullopt;

121/// @brief Constrains what may be used to single out one of several foreign keys into the same table.
122///
123/// Two forms are accepted:
124/// - `std::nullopt` (`AutoDetectRelation`) - resolve automatically; ambiguity is a compile error.
125/// - a `SqlRealName` (or anything convertible to `std::string_view`) - the SQL column name of the
126/// foreign key to use.
127///
128/// A relationship selector must stay inert at *declaration* time: the two records of a relationship
129/// reference each other, so neither is complete where the other is declared. That rules out
130/// pointer-to-member selectors and is why the foreign key is named by its column.
131///
132/// @ingroup DataMapper
133template <auto Selector>
134concept RelationSelector =
135 std::same_as<std::remove_cvref_t<decltype(Selector)>, std::nullopt_t> || requires { std::string_view { Selector }; };

140 /// @brief Tests whether member @p I of @p Record is singled out by @p Selector.
141 ///
142 /// @tparam Selector The relationship selector, see the RelationSelector concept.
143 /// @tparam I Member index within @p Record.
144 /// @tparam Record The record holding the foreign key.
145 /// @return `true` when the selector accepts the member.
146 template <auto Selector, size_t I, typename Record>
147 constexpr bool RelationSelectorMatches()
148 {
149 if constexpr (std::same_as<std::remove_cvref_t<decltype(Selector)>, std::nullopt_t>)
150 return true;
151 else
152 return Lightweight::FieldNameAt<I, Record> == std::string_view { Selector };
153 }

155 /// @brief Outcome of scanning a child record for the `BelongsTo` members pointing back to an owner record.
156 struct InverseBelongsToLookup
157 {
158 /// Member index of the first matching `BelongsTo`, or `RecordMemberCount` if there is none.
159 size_t index {};
160
161 /// Number of matching `BelongsTo` members found.
162 size_t count {};
163
164 /// Number of `BelongsTo` members referencing the owner record, before the selector was applied.
165 size_t candidates {};
166 };
167
168 /// @brief Scans @p ChildRecord for the `BelongsTo` members whose referenced record is @p OwnerRecord.
169 ///
170 /// @tparam OwnerRecord The record on the "one" side of a one-to-many relationship.
171 /// @tparam ChildRecord The record on the "many" side, holding the foreign key.
172 /// @tparam Selector Singles out one of several foreign keys, see the RelationSelector concept.
173 /// @return The index of the first match, how many matches exist, and how many candidates the
174 /// selector had to choose from.
175 template <typename OwnerRecord, typename ChildRecord, auto Selector = AutoDetectRelation>
176 constexpr InverseBelongsToLookup FindInverseBelongsTo()
177 {
178 return FoldRecordMembers<ChildRecord>(
179 InverseBelongsToLookup { .index = RecordMemberCount<ChildRecord>, .count = 0, .candidates = 0 },
180 []<size_t I, typename MemberType>(InverseBelongsToLookup const accum) constexpr -> InverseBelongsToLookup {
181 // The two conditions must nest: `MemberType::ReferencedRecord` does not exist on plain
182 // fields, and `&&` inside a single `if constexpr` would still instantiate it.
183 if constexpr (IsBelongsTo<MemberType>)
184 {
185 if constexpr (std::same_as<typename MemberType::ReferencedRecord, OwnerRecord>)
186 {
187 if constexpr (RelationSelectorMatches<Selector, I, ChildRecord>())
188 return { .index = accum.count == 0 ? I : accum.index,
189 .count = accum.count + 1,
190 .candidates = accum.candidates + 1 };
191 else
192 return { .index = accum.index, .count = accum.count, .candidates = accum.candidates + 1 };
193 }
194 else
195 return accum;
196 }
197 else
198 return accum;
199 });
200 }

202 /// @brief Resolves - and validates - the inverse `BelongsTo` member of a relationship.
203 ///
204 /// Instantiating this template fails to compile unless @p ChildRecord declares exactly one
205 /// `BelongsTo` member that references @p OwnerRecord and is accepted by @p Selector. The compiler's
206 /// instantiation backtrace names the record types involved.
207 ///
208 /// @tparam OwnerRecord The record being referenced (the "one" side of the relationship).
209 /// @tparam ChildRecord The record holding the foreign key (the "many" side).
210 /// @tparam Selector Singles out one of several foreign keys, see the RelationSelector concept.
211 template <typename OwnerRecord, typename ChildRecord, auto Selector = AutoDetectRelation>
212 requires RelationSelector<Selector>
213 struct InverseBelongsToResolver
214 {
215 /// The raw lookup result for @p OwnerRecord within @p ChildRecord.
216 static constexpr InverseBelongsToLookup Lookup = FindInverseBelongsTo<OwnerRecord, ChildRecord, Selector>();
217
218 static_assert(Lookup.candidates != 0,
219 "This relationship requires the referencing record to declare a BelongsTo member pointing at "
220 "the referenced record's primary key. No such member was found. "
221 "See the instantiation backtrace below for the two record types involved.");
222
223 static_assert(Lookup.candidates == 0 || Lookup.count != 0,
224 "No BelongsTo member matches the foreign key column named by this relationship. The referencing "
225 "record does declare a BelongsTo pointing at the referenced record, but none of them uses that "
226 "column name - check the SqlRealName spelling on both sides. "
227 "See the instantiation backtrace below for the two record types involved.");
228
229 static_assert(Lookup.count <= 1,
230 "This relationship is ambiguous: the referencing record declares more than one BelongsTo member "
231 "pointing at the referenced record's primary key. Name the foreign key column to disambiguate, "
232 "e.g. HasMany<Child, SqlRealName { \"owner_id\" }>. "
233 "See the instantiation backtrace below for the two record types involved.");
234
235 /// Member index of the inverse `BelongsTo` inside @p ChildRecord.
236 /// Clamped to 0 on failure so that the static_asserts above are the only diagnostics emitted.
237 static constexpr size_t Index = Lookup.count == 1 ? Lookup.index : 0;
238 };

243///
244/// This is how `HasMany`, `HasManyThrough` and `HasOneThrough` locate their foreign key column: by matching
245/// the relationship *type*, never by member position. Using it is a hard compile-time error when

294/// @brief Requires that T maps onto a column of its record's table.
295///
296/// This is satisfied by fields with storage (Field, BelongsTo) as well as by plain record members
297/// that are directly bindable as an output column (a record may be a plain struct of bindable
298/// members). Relation members (HasMany, HasManyThrough, HasOneThrough, ...) have no column of their
299/// own and therefore must be skipped by every column-enumerating code path, such as the projection
300/// of a SELECT statement.
301///
302/// @ingroup DataMapper
303template <typename T>
304concept RecordColumnMember = FieldWithStorage<T> || SqlOutputColumnBinder<T>;
305
306/// @brief Represents the number of members of a record that map onto a column of a result set.
307///
308/// This is the width the record occupies in a projection built from the @c RecordColumnMember
309/// concept (e.g. @c SqlSelectQueryBuilder::Fields), and therefore the amount by which the index must be
310/// advanced to reach the first column of the record that follows it in a multi-record projection.
311/// It differs from @c RecordMemberCount exactly by the number of relation members (HasMany,
312/// HasManyThrough, HasOneThrough, ...), which have no column of their own.
313///
314/// @ingroup DataMapper
315template <typename Record>
316constexpr size_t RecordColumnCount =
317 FoldRecordMembers<Record>(size_t { 0 }, []<size_t I, typename Field>(size_t const accum) constexpr {
318 if constexpr (RecordColumnMember<Field>)
319 return accum + 1;
320 else
321 return accum;
322 });

374 /// Collects the value types of every `PrimaryKey` member of @p Record, in declaration order.
375 template <typename Record>
376 struct RecordPrimaryKeyTupleHelper
377 {

380 using FieldType = RecordMemberTypeOf<J, Record>;

401 /// Number of members of @p Record declared as `PrimaryKey::AutoAssign`.
402 ///
403 /// Auto-assignment yields a single value that is then written into every primary key member, so more
404 /// than one such member cannot be honoured - see the static_assert in
405 /// `DataMapper::GenerateAutoAssignPrimaryKey`.
406 /// Whether `GenerateAutoAssignPrimaryKey` actually produces a value for @p FieldType.
407 ///
408 /// `PrimaryKey::AutoAssign` only generates for a GUID or an incrementable value; on any other type
409 /// (a string key, say) it silently generates nothing and the caller supplies the value. Only the
410 /// generating case can collide across several key members, so only it is counted.
411 template <typename ValueType>
412 concept IncrementableKeyValue = requires(ValueType value) { value + 1; };
413
414 /// Whether @p ValueType is one of the two kinds `GenerateAutoAssignPrimaryKey` actually generates a
415 /// value for: a GUID (via `SqlGuid::Create()`) or an incrementable value (via `MAX(...) + 1`).
416 template <typename ValueType>
417 concept AutoAssignableKeyValue = std::same_as<ValueType, SqlGuid> || IncrementableKeyValue<ValueType>;
418
419 template <typename FieldType>
420 concept GeneratesAutoAssignedKey = IsField<FieldType> && IsAutoAssignPrimaryKeyField<FieldType>::value
421 && AutoAssignableKeyValue<typename FieldType::ValueType>;
422
423 template <typename Record>
424 constexpr std::size_t AutoAssignPrimaryKeyFieldCount =
425 FoldRecordMembers<Record>(std::size_t { 0 }, []<std::size_t I, typename FieldType>(std::size_t const accum) {
426 if constexpr (GeneratesAutoAssignedKey<FieldType>)
427 return accum + 1;
428 else
429 return accum;
430 });
431} // namespace detail

460/// @brief Reads every primary key value of @p record, in member declaration order.
461///
462/// This is the order a primary key lookup binds its arguments in, so the returned tuple can be applied
463/// straight to `QuerySingle`/`Update`/`Delete`.
464///
465/// @param record Record to read.
466/// @return The key values as a tuple.
467///
468/// @ingroup DataMapper
469template <typename Record>
470[[nodiscard]] RecordPrimaryKeyTuple<Record> GetPrimaryKeyFields(Record const& record)
471{
472 // Mirrors RecordPrimaryKeyTupleHelper's compile-time tuple_cat construction (one std::tuple<> or
473 // std::tuple<ValueType> per member, concatenated), but reads each primary-key member's value instead
474 // of just its type. The two therefore cannot disagree on which members are collected or in what
475 // order, and no runtime index-matching against the heterogeneous tuple is needed.
476 return []<std::size_t... I>(Record const& record, std::index_sequence<I...>) {
477 return std::tuple_cat([&record]<std::size_t J>() {
478 using FieldType = RecordMemberTypeOf<J, Record>;
479 if constexpr (IsField<FieldType>)
480 {
481 if constexpr (FieldType::IsPrimaryKey)
482 return std::tuple<typename FieldType::ValueType> { GetRecordMemberAt<J>(record).Value() };
483 else
484 return std::tuple<> {};
485 }
486 else
487 return std::tuple<> {};
488 }.template operator()<I>()...);
489 }(record, std::make_index_sequence<RecordMemberCount<Record>> {});
490}
