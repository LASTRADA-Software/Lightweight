23/// @brief This HasMany<OtherRecord> represents a simple one-to-many relationship between two records.
24///
25/// The HasMany<OtherRecord> is a member of the "one" side of the relationship.
26///
27/// `OtherRecord` must declare exactly one `BelongsTo` member that points back to this "one" side.
28/// That member is located by matching the relationship *type*, not by its position in either record,
29/// so the two relationship members may be declared at any index. Declaring no such `BelongsTo`, or
30/// more than one (which would make the inverse ambiguous), is a compile-time error.
31///
32/// @see InverseBelongsToIndexOf
33///
34/// @see DataMapper, Field, HasManyThrough
35/// @ingroup DataMapper

121 struct Loader
122 {
123 std::function<size_t()> count {};
124 std::function<ReferencedRecordList()> all {};
125 std::function<void(std::function<void(ReferencedRecord const&)>)> each {};
126
127 std::weak_ordering operator<=>(Loader const& /*other*/) const noexcept
128 {
129 return std::weak_ordering::equivalent; // Loader is not comparable, so we return equivalent
130 }
131 };
