48/// Determines if the given SQL error is a transient error that can be retried.
49///
50/// Thin adapter over @ref Lightweight::GenericRetryOps(), which is where the classification
51/// actually lives — the backup engine no longer carries its own copy of the heuristics. The
52/// dialect-agnostic classifier is used because this helper is reached from contexts that have an
53/// error but not the connection it came from; a caller that does have a connection should build a
54/// @ref Lightweight::SqlRetryPolicy with @c SqlRetryPolicy::For() and get the sharper per-DBMS
55/// classification instead.
56///
57/// Transient errors include:
58/// - Connection errors (ODBC class 08)
59/// - Timeout errors (HYT00, HYT01)
60/// - Transaction rollback due to deadlock/serialization (class 40)
61/// - Database locked (common in SQLite)
62///
63/// @param error The SQL error information to check.
64/// @return true if the error is transient and the operation can be retried.
65LIGHTWEIGHT_API bool IsTransientError(SqlErrorInfo const& error);

144/// Retries a function on transient errors with exponential backoff.
145///
146/// A thin wrapper over @ref Lightweight::SqlRetryPolicy::Execute() that routes each retry notice
147/// into @p progress. @p func is taken by value because it is invoked repeatedly — forwarding an
148/// rvalue more than once would use a moved-from callable.
149///
150/// @tparam Func The callable type.
151/// @param func The function to execute.
152/// @param settings Retry configuration.
153/// @param progress Progress manager for reporting retry attempts.
154/// @param operation Name of the operation for progress messages.
155/// @return The result of the function.
156/// @throws SqlException if max retries exceeded or non-transient error occurs.
157template <typename Func>
158auto RetryOnTransientError(Func func, RetrySettings const& settings, ProgressManager& progress, std::string const& operation)
159 -> decltype(func())
160{
161 auto policy = SqlRetryPolicy { settings };
162
163 policy.SetRetryObserver([&progress, &operation, &settings](SqlRetryAttempt const& attempt) {
164 progress.Update({ .state = Progress::State::Warning,
165 .tableName = operation,
166 .currentRows = 0,
167 .totalRows = std::nullopt,
168 // Formatting the error info reproduces SqlException::what() exactly, so
169 // the wording of these progress lines is unchanged by the migration.
170 .message = std::format("Transient error, retry {}/{}: {}",
171 attempt.retryNumber,
172 settings.maxRetries,
173 std::format("{}", attempt.error)) });
174 });
175
176 return policy.Execute(std::move(func));
177}

184/// Reads a ZIP entry into a container.
185///
186/// @tparam Container The container type (e.g., std::string, std::vector<uint8_t>).
187/// @param zip The ZIP archive handle.
188/// @param index The index of the entry to read.
189/// @param size The size of the entry in bytes.
190/// @return The container with the entry contents, or empty on failure.
191template <typename Container>
192Container ReadZipEntry(zip_t* zip, zip_int64_t index, zip_uint64_t size)
193{
194 zip_file_t* file = zip_fopen_index(zip, static_cast<zip_uint64_t>(index), 0);
195 if (!file)
196 return {}; // LCOV_EXCL_LINE - zip file open failure
197
198 Container data;
199 data.resize(size);
200
201 zip_int64_t bytesRead = zip_fread(file, data.data(), size);
202 zip_fclose(file);
203
204 if (bytesRead < 0 || std::cmp_not_equal(bytesRead, size))
205 return {}; // LCOV_EXCL_LINE - zip file read failure
206
207 return data;
208}
