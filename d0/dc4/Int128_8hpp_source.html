26 /// Signed 128-bit integer with just enough arithmetic to carry a fixed-point unscaled value.
27 ///
28 /// This exists so that `SqlNumeric`'s unscaled carrier is 128 bits wide on *every* toolchain. Where
29 /// the compiler offers `__int128_t` that type is used directly (see `Int128` below); MSVC and
30 /// clang-cl have no such type, and this software implementation stands in for it there. Without it
31 /// the carrier would fall back to `int64_t`, which holds only 18 decimal digits, and the widest
32 /// declarable precision would differ between toolchains — meaning a `ddl2cpp`-generated record for a
33 /// `money` column (`DECIMAL(19, 4)`) would compile under GCC/Clang and fail under MSVC. A schema is a
34 /// property of the database, not of the compiler that happens to build the client.
35 ///
36 /// The operation set is deliberately minimal — construction from and conversion to the floating-point
37 /// types, negation, comparison, and decimal rendering — because that is all `SqlNumeric` needs. It is
38 /// not a general-purpose big integer: there is no multiplication, no division by another `Int128Soft`,
39 /// and no bitwise interface beyond what the conversions require.
40 ///
41 /// Representation is sign-magnitude-free two's complement across a low/high pair, matching the
42 /// little-endian layout of `SQL_NUMERIC_STRUCT::val` so that a `std::memcpy` in either direction is a
43 /// faithful round-trip.
44 struct Int128Soft
45 {
46 /// Low 64 bits of the two's-complement magnitude.
47 std::uint64_t low {};
48
49 /// High 64 bits of the two's-complement magnitude, including the sign bit.
50 std::uint64_t high {};
51
52 /// Default-constructs a zero value.
53 constexpr Int128Soft() noexcept = default;
54
55 /// Constructs from a signed 64-bit integer, sign-extending into the high word.
56 /// @param value Value to widen.
57 constexpr Int128Soft(std::int64_t value) noexcept: // NOLINT(google-explicit-constructor)
58 low { static_cast<std::uint64_t>(value) },
59 high { value < 0 ? ~std::uint64_t { 0 } : std::uint64_t { 0 } }
60 {
61 }
62
63 /// Constructs from an unsigned 64-bit integer, zero-extending into the high word.
64 /// @param value Value to widen.
65 constexpr Int128Soft(std::uint64_t value) noexcept: // NOLINT(google-explicit-constructor)
66 low { value }
67 {
68 }
69
70 /// Constructs from an explicit low/high word pair.
71 /// @param lowWord Low 64 bits.
72 /// @param highWord High 64 bits, including the sign bit.
73 constexpr Int128Soft(std::uint64_t lowWord, std::uint64_t highWord) noexcept:
74 low { lowWord },
75 high { highWord }
76 {
77 }

79 /// Constructs from a floating-point value by truncation toward zero.
80 ///
81 /// Values whose magnitude is at or beyond 2^127 saturate to the corresponding extreme rather than
82 /// invoking the undefined behaviour an out-of-range `static_cast` to an integer type would. That
83 /// is the whole point of routing through this type: the `int64_t` fallback it replaces silently
84 /// flipped the sign of `money`'s maximum on x86-64.
85 ///
86 /// @param value Value to truncate. Must be finite; NaN yields zero.
87 template <std::floating_point T>
88 constexpr explicit Int128Soft(T value) noexcept
89 {
90 if (!(value == value)) // NaN — no meaningful integer image.
91 return;
92
93 auto const negative = value < T { 0 };
94 auto magnitude = negative ? -value : value;
95
96 // 2^127, the first magnitude a signed 128-bit integer cannot represent. Computed by repeated
97 // doubling so it stays exact in every floating-point format (2^127 is a power of two, hence
98 // representable in all of them, but a literal would depend on the type's width).
99 auto limit = T { 1 };
100 for (auto bit = 0; bit != 127; ++bit)
101 limit *= T { 2 };
102
103 if (magnitude >= limit)
104 {
105 // Saturate: -2^127 .. 2^127-1.
106 low = negative ? std::uint64_t { 0 } : ~std::uint64_t { 0 };
107 high = negative ? (std::uint64_t { 1 } << 63) : (~std::uint64_t { 0 } >> 1);
108 return;
109 }
110
111 // Split into two 64-bit halves. `magnitude / 2^64` truncated is the high word; the remainder
112 // is the low word. Both subexpressions are exact: the quotient is < 2^63 and the remainder is
113 // recovered by subtraction rather than by a second division.
114 auto divisor = T { 1 };
115 for (auto bit = 0; bit != 64; ++bit)
116 divisor *= T { 2 };
117
118 auto const highPart = std::trunc(magnitude / divisor);
119 auto const lowPart = std::trunc(magnitude - (highPart * divisor));
120
121 low = static_cast<std::uint64_t>(lowPart);
122 high = static_cast<std::uint64_t>(highPart);
123
124 if (negative)
125 *this = -*this;
126 }
127
128 /// @return `true` if the value is negative.
129 [[nodiscard]] constexpr bool IsNegative() const noexcept
130 {
131 return (high >> 63) != 0;
132 }
133
134 /// Negates the value (two's complement).
135 /// @return The additive inverse. `-(-2^127)` saturates to itself, as it does for every two's-complement type.
136 [[nodiscard]] constexpr Int128Soft operator-() const noexcept
137 {
138 auto const invertedLow = ~low;
139 auto const invertedHigh = ~high;
140 auto const carriedLow = invertedLow + 1;
141 auto const carriedHigh = invertedHigh + (carriedLow < invertedLow ? 1 : 0);
142 return { carriedLow, carriedHigh };
143 }
144
145 /// @return The value unchanged (provided for symmetry with `operator-`).
146 [[nodiscard]] constexpr Int128Soft operator+() const noexcept
147 {
148 return *this;
149 }
150
151 /// Converts to a floating-point type.
152 ///
153 /// Precision is bounded by the target type's significand, exactly as it is for the native
154 /// `__int128_t` this stands in for.
155 ///
156 /// @tparam T Target floating-point type.
157 /// @return The value as `T`, rounded to that type's precision.
158 template <std::floating_point T>
159 [[nodiscard]] constexpr T To() const noexcept
160 {
161 auto const negative = IsNegative();
162 auto const magnitude = negative ? -*this : *this;
163
164 auto scale = T { 1 };
165 for (auto bit = 0; bit != 64; ++bit)
166 scale *= T { 2 };
167
168 auto const result = (static_cast<T>(magnitude.high) * scale) + static_cast<T>(magnitude.low);
169 return negative ? -result : result;
170 }
171
172 /// @return The value as a `float`.
173 [[nodiscard]] constexpr explicit operator float() const noexcept
174 {
175 return To<float>();
176 }
177
178 /// @return The value as a `double`.
179 [[nodiscard]] constexpr explicit operator double() const noexcept
180 {
181 return To<double>();
182 }
183
184 /// @return The value as a `long double`.
185 [[nodiscard]] constexpr explicit operator long double() const noexcept
186 {
187 return To<long double>();
188 }
189
190 /// Narrows to a signed 64-bit integer, truncating the high word.
191 /// @return The low 64 bits reinterpreted as signed, matching `static_cast<int64_t>` on a native `__int128_t`.
192 [[nodiscard]] constexpr explicit operator std::int64_t() const noexcept
193 {
194 return static_cast<std::int64_t>(low);
195 }
196
197 /// Narrows to an unsigned 64-bit integer, truncating the high word.
198 /// @return The low 64 bits, matching `static_cast<uint64_t>` on a native `__int128_t`.
199 [[nodiscard]] constexpr explicit operator std::uint64_t() const noexcept
200 {
201 return low;
202 }
203
204 /// Narrows to a signed 32-bit integer, truncating.
205 [[nodiscard]] constexpr explicit operator std::int32_t() const noexcept
206 {
207 return static_cast<std::int32_t>(low);
208 }
209
210 /// Equality comparison.
211 [[nodiscard]] constexpr bool operator==(Int128Soft const& other) const noexcept = default;
212
213 /// Three-way comparison, signed.
214 /// @param other Value to compare against.
215 /// @return The ordering of `*this` relative to `other`.
216 [[nodiscard]] constexpr std::strong_ordering operator<=>(Int128Soft const& other) const noexcept
217 {
218 if (auto const negative = IsNegative(); negative != other.IsNegative())
219 return negative ? std::strong_ordering::less : std::strong_ordering::greater;
220
221 if (high != other.high)
222 return high <=> other.high;
223
224 return low <=> other.low;
225 }
226 };

228 /// Renders a software 128-bit integer in base 10.
229 ///
230 /// `std::format` has no support for 128-bit integers — not even for the native `__int128_t` — so
231 /// decimal rendering is done here by schoolbook long division over 32-bit limbs.
232 ///
233 /// @param value Value to render.
234 /// @return Decimal representation, with a leading '-' for negative values.
235 [[nodiscard]] inline std::string ToDecimalString(Int128Soft value) noexcept
236 {
237 if (value == Int128Soft {})
238 return "0";
239
240 auto const negative = value.IsNegative();
241
242 // Work on the magnitude as an unsigned 128-bit pair. Negating the most-negative value overflows
243 // back to itself, but its two's-complement bit pattern is exactly the unsigned magnitude 2^127,
244 // so treating the words as unsigned below is correct for it too.
245 auto const magnitude = negative ? -value : value;
246 auto low = magnitude.low;
247 auto high = magnitude.high;
248
249 // Divide repeatedly by 10^9, peeling off nine digits per pass. The divisor has to stay below
250 // 2^32: each step of the long division forms `(remainder << 32) | limb`, and that only fits a
251 // std::uint64_t while `remainder < divisor <= 2^32`. (10^19 would be the largest power of ten a
252 // std::uint64_t *holds*, but its remainders overflow that shift — the digits then come out
253 // scrambled, which is exactly what the round-trip tests catch.)
254 constexpr auto chunkDivisor = std::uint64_t { 1'000'000'000ULL };
255 constexpr auto chunkDigits = 9;
256
257 auto chunks = std::string {};
258
259 while (high != 0 || low != 0)
260 {
261 auto remainder = std::uint64_t { 0 };
262
263 auto const limbs = std::array<std::uint32_t, 4> {
264 static_cast<std::uint32_t>(high >> 32),
265 static_cast<std::uint32_t>(high & 0xFFFF'FFFFULL),
266 static_cast<std::uint32_t>(low >> 32),
267 static_cast<std::uint32_t>(low & 0xFFFF'FFFFULL),
268 };
269
270 auto quotientLimbs = std::array<std::uint32_t, 4> {};
271
272 for (auto index = std::size_t { 0 }; index != limbs.size(); ++index)
273 {
274 // `remainder < 10^9 < 2^32`, so `(remainder << 32) | limb` stays within 64 bits.
275 auto const dividend = (remainder << 32) | limbs[index];
276 quotientLimbs[index] = static_cast<std::uint32_t>(dividend / chunkDivisor);
277 remainder = dividend % chunkDivisor;
278 }
279
280 high = (static_cast<std::uint64_t>(quotientLimbs[0]) << 32) | quotientLimbs[1];
281 low = (static_cast<std::uint64_t>(quotientLimbs[2]) << 32) | quotientLimbs[3];
282
283 // Emit this chunk's digits least-significant first; they are zero-padded unless this was the
284 // final (most significant) chunk, whose padding is trimmed after the loop.
285 for (auto digit = 0; digit != chunkDigits; ++digit)
286 {
287 chunks.push_back(static_cast<char>('0' + static_cast<char>(remainder % 10)));
288 remainder /= 10;
289 }
290 }
291
292 // Drop the zero padding of the most significant chunk, then reverse into print order.
293 while (chunks.size() > 1 && chunks.back() == '0')
294 chunks.pop_back();
295
296 if (negative)
297 chunks.push_back('-');
298
299 return std::string { chunks.rbegin(), chunks.rend() };
300 }

304/// The unscaled carrier `SqlNumeric` keeps `value * 10^Scale` in: a signed 128-bit integer on every
305/// supported toolchain.
306///
307/// Where the compiler provides `__int128_t` that is used directly, so nothing changes for GCC and
308/// Clang. MSVC and clang-cl get `detail::Int128Soft`, a software stand-in with the same width and the
309/// same conversion behaviour. Both are 16 bytes and little-endian, matching `SQL_NUMERIC_STRUCT::val`
310/// byte for byte.
311#if defined(LIGHTWEIGHT_HAVE_NATIVE_INT128)
312using Int128 = __int128_t;
313#else
314using Int128 = detail::Int128Soft;
315#endif
316
317static_assert(sizeof(Int128) == 16, "The unscaled carrier must be exactly as wide as SQL_NUMERIC_STRUCT::val.");

322 /// Renders the unscaled carrier in base 10, whichever of the two implementations it is.
323 ///
324 /// `std::format` supports neither `__int128_t` nor `Int128Soft`, so both paths are handled here.
325 ///
326 /// @param value Value to render.
327 /// @return Decimal representation, with a leading '-' for negative values.
328 [[nodiscard]] inline std::string Int128ToString(Int128 value) noexcept
329 {
330#if defined(LIGHTWEIGHT_HAVE_NATIVE_INT128)
331 auto const negative = value < 0;
332 auto magnitude = static_cast<__uint128_t>(negative ? -value : value);
333
334 if (magnitude == 0)
335 return "0";
336
337 auto reversed = std::string {};
338 while (magnitude != 0)
339 {
340 reversed.push_back(static_cast<char>('0' + static_cast<char>(magnitude % 10)));
341 magnitude /= 10;
342 }
343 if (negative)
344 reversed.push_back('-');
345
346 return std::string { reversed.rbegin(), reversed.rend() };
347#else
348 return ToDecimalString(value);
349#endif
350 }
