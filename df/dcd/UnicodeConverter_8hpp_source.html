24 /// U+FFFD REPLACEMENT CHARACTER: what every ill-formed code unit and every code point that is no Unicode
25 /// scalar value converts to, one per unit, so that no well-formed unit next to it is lost.
26 inline constexpr auto ReplacementCharacter = char32_t { 0xFFFD };

31 template <>
32 struct LIGHTWEIGHT_API UnicodeConverter<char8_t>
33 {
34 // Converts a UTF-32 code point to one to four UTF-8 code units. A surrogate or a code point above
35 // U+10FFFF has no UTF-8 form and becomes U+FFFD.
36 template <typename OutputIterator>
37 static constexpr OutputIterator Convert(char32_t input, OutputIterator output) noexcept
38 {
39 if (input <= 0x7F)
40 {
41 *output++ = static_cast<char8_t>(input & 0b0111'1111);
42 }
43 else if (input <= 0x07FF)
44 {
45 *output++ = static_cast<char8_t>(((input >> 6) & 0b0001'1111) | 0b1100'0000);
46 *output++ = static_cast<char8_t>(((input >> 0) & 0b0011'1111) | 0b1000'0000);
47 }
48 else if (input <= 0xFFFF)
49 {
50 if (input >= 0xD800 && input <= 0xDFFF)
51 input = ReplacementCharacter;
52 *output++ = static_cast<char8_t>(((input >> 12) & 0b0000'1111) | 0b1110'0000);
53 *output++ = static_cast<char8_t>(((input >> 6) & 0b0011'1111) | 0b1000'0000);
54 *output++ = static_cast<char8_t>(((input >> 0) & 0b0011'1111) | 0b1000'0000);
55 }
56 else if (input > 0x10FFFF)
57 {
58 return Convert(ReplacementCharacter, output);
59 }
60 else
61 {
62 *output++ = static_cast<char8_t>(((input >> 18) & 0b0000'0111) | 0b1111'0000);
63 *output++ = static_cast<char8_t>(((input >> 12) & 0b0011'1111) | 0b1000'0000);
64 *output++ = static_cast<char8_t>(((input >> 6) & 0b0011'1111) | 0b1000'0000);
65 *output++ = static_cast<char8_t>(((input >> 0) & 0b0011'1111) | 0b1000'0000);
66 }
67 return output;
68 }
69 };

71 template <>
72 struct LIGHTWEIGHT_API UnicodeConverter<char16_t>
73 {
74 // Converts a UTF-32 code point to one or two UTF-16 code units. A surrogate or a code point above
75 // U+10FFFF has no UTF-16 form and becomes U+FFFD.
76 template <typename OutputIterator>
77 static constexpr OutputIterator Convert(char32_t input, OutputIterator output) noexcept
78 {
79 if (input < 0xD800) // [0x0000 .. 0xD7FF]
80 {
81 *output++ = char16_t(input);
82 return output;
83 }
84 else if (input < 0x10000)
85 {
86 // A surrogate code point is no scalar value; written as is it would pair up with a neighbour.
87 *output++ = char16_t(input < 0xE000 ? ReplacementCharacter : input);
88 return output;
89 }
90 else if (input < 0x110000) // [0xD800 .. 0xDBFF] [0xDC00 .. 0xDFFF]
91 {
92 *output++ = char16_t(0xD7C0 + (input >> 10));
93 *output++ = char16_t(0xDC00 + (input & 0x3FF));
94 return output;
95 }
96 else
97 {
98 *output++ = char16_t(ReplacementCharacter); // Too large for UTF-16.
99 return output;
100 }
101 }
102 };

104 /// Decodes UTF-8 one code point at a time (Unicode 15, table 3-7, "well-formed UTF-8 byte sequences").
105 ///
106 /// An ill-formed sequence gives one U+FFFD for its maximal subpart, Unicode's recommended practice (section
107 /// 3.9, "U+FFFD Substitution of Maximal Subparts") and what WHATWG's decoder produces (MultiByteToWideChar(CP_UTF8)
108 /// agrees on truncated sequences and stray bytes, but takes a lead byte and an out-of-range second byte as one):
109 /// a lead byte and the continuation bytes that can still begin a well-formed sequence after it (E2 82 41
110 /// gives U+FFFD A), or a single byte that begins none (an overlong lead, an encoded surrogate's second byte,
111 /// a stray continuation byte). Decoding resumes at the byte that ended the subpart, so no valid byte is lost.
112 struct Utf32Converter
113 {
114 /// The code point an ill-formed byte decodes to.
115 static constexpr auto InvalidCodePoint = ReplacementCharacter;
116
117 /// One decoded code point and the bytes it took.
118 struct Decoded
119 {
120 char32_t codePoint; //!< The code point, or InvalidCodePoint.
121 std::size_t length; //!< Bytes consumed: the sequence's length, or the maximal subpart's if ill-formed.
122 };
123
124 /// The well-formed multi-byte sequences, by lead byte: how many bytes they take, and the range of the
125 /// second byte, which rules out overlong encodings, surrogates and code points above U+10FFFF.
126 struct SequenceRule
127 {
128 char8_t firstLead; //!< The first lead byte the rule covers.
129 char8_t lastLead; //!< The last lead byte the rule covers.
130 std::uint8_t length; //!< The sequence's length in bytes, lead byte included.
131 char8_t secondMin; //!< The smallest second byte of a well-formed sequence.
132 char8_t secondMax; //!< The largest second byte of a well-formed sequence.
133 };
134
135 static constexpr auto SequenceRules = std::array {
136 SequenceRule { .firstLead = 0xC2, .lastLead = 0xDF, .length = 2, .secondMin = 0x80, .secondMax = 0xBF },
137 SequenceRule { .firstLead = 0xE0, .lastLead = 0xE0, .length = 3, .secondMin = 0xA0, .secondMax = 0xBF },
138 SequenceRule { .firstLead = 0xE1, .lastLead = 0xEC, .length = 3, .secondMin = 0x80, .secondMax = 0xBF },
139 SequenceRule { .firstLead = 0xED, .lastLead = 0xED, .length = 3, .secondMin = 0x80, .secondMax = 0x9F },
140 SequenceRule { .firstLead = 0xEE, .lastLead = 0xEF, .length = 3, .secondMin = 0x80, .secondMax = 0xBF },
141 SequenceRule { .firstLead = 0xF0, .lastLead = 0xF0, .length = 4, .secondMin = 0x90, .secondMax = 0xBF },
142 SequenceRule { .firstLead = 0xF1, .lastLead = 0xF3, .length = 4, .secondMin = 0x80, .secondMax = 0xBF },
143 SequenceRule { .firstLead = 0xF4, .lastLead = 0xF4, .length = 4, .secondMin = 0x80, .secondMax = 0x8F },
144 };
145
146 /// The rule of the sequences lead byte @p lead begins.
147 /// @param lead The lead byte.
148 /// @return The rule, or @c nullptr for a byte that begins no well-formed multi-byte sequence.
149 static constexpr SequenceRule const* FindSequenceRule(char8_t lead) noexcept
150 {
151 for (auto const& rule: SequenceRules)
152 if (lead >= rule.firstLead && lead <= rule.lastLead)
153 return &rule;
154 return nullptr;
155 }
156
157 /// Decodes the code point @p text starts with.
158 /// @param text The input; not empty.
159 /// @return The code point and the bytes it took.
160 static constexpr Decoded Decode(std::u8string_view text) noexcept
161 {
162 auto const lead = text.front();
163 if (lead < 0x80)
164 return { .codePoint = static_cast<char32_t>(lead), .length = 1 };
165
166 SequenceRule const* const rule = FindSequenceRule(lead);
167 // A byte that begins no well-formed sequence, or a lead byte whose second byte cannot follow it, is a
168 // maximal subpart of its own.
169 if (rule == nullptr || text.size() < 2 || text[1] < rule->secondMin || text[1] > rule->secondMax)
170 return { .codePoint = InvalidCodePoint, .length = 1 };
171
172 // The lead byte keeps 7 - length payload bits; each continuation byte six. Past the second byte, any
173 // continuation byte can follow; the first that is missing or is none ends the subpart before it.
174 auto codePoint = static_cast<char32_t>(lead & (0b0111'1111 >> rule->length));
175 for (auto const index: std::views::iota(std::size_t { 1 }, std::size_t { rule->length }))
176 {
177 if (index >= text.size() || (text[index] & 0b1100'0000) != 0b1000'0000)
178 return { .codePoint = InvalidCodePoint, .length = index };
179 codePoint = (codePoint << 6) | (text[index] & 0b0011'1111);
180 }
181 return { .codePoint = codePoint, .length = rule->length };
182 }
183 };

185 /// Decodes UTF-16 one code point at a time. An unpaired surrogate (a high one not followed by a low one,
186 /// or a low one not preceded by a high one) gives U+FFFD and decoding resumes at the next code unit, so
187 /// the unit after it is decoded on its own and never swallowed.
188 struct Utf16Decoder
189 {
190 /// One decoded code point and the code units it took.
191 struct Decoded
192 {
193 char32_t codePoint; //!< The code point, or ReplacementCharacter.
194 std::size_t length; //!< Code units consumed: 2 for a surrogate pair, else 1.
195 };
196
197 /// Decodes the code point @p text starts with.
198 /// @param text The input; not empty.
199 /// @return The code point and the code units it took.
200 [[nodiscard]] static constexpr Decoded Decode(std::u16string_view text) noexcept
201 {
202 auto const first = text.front();
203 if (first < 0xD800 || first > 0xDFFF)
204 return { .codePoint = first, .length = 1 };
205 if (first <= 0xDBFF && text.size() > 1 && text[1] >= 0xDC00 && text[1] <= 0xDFFF)
206 return { .codePoint = 0x10000 + ((static_cast<char32_t>(first) - 0xD800) << 10)
207 + (static_cast<char32_t>(text[1]) - 0xDC00),
208 .length = 2 };
209 return { .codePoint = ReplacementCharacter, .length = 1 };
210 }
211 };
212
213 /// The code points of a UTF-8 string (see Utf32Converter::Decode()).
214 struct [[nodiscard]] Utf32Iterator
215 {
216 std::u8string_view u8InputString;
217
218 struct [[nodiscard]] iterator
219 {
220 std::u8string_view rest {}; //!< The input from the current code point on.
221
222 constexpr char32_t operator*() const noexcept
223 {
224 return Utf32Converter::Decode(rest).codePoint;
225 }
226
227 constexpr iterator& operator++() noexcept
228 {
229 rest.remove_prefix(Utf32Converter::Decode(rest).length);
230 return *this;
231 }
232
233 constexpr iterator operator++(int) noexcept
234 {
235 auto const previous = *this;
236 ++*this;
237 return previous;
238 }
239
240 /// Positions compare equal: the end is where the input ends, whatever was decoded last.
241 constexpr bool operator==(iterator const& other) const noexcept
242 {
243 return rest.data() == other.rest.data();
244 }
245 };
246
247 iterator begin() const noexcept
248 {
249 return iterator { .rest = u8InputString };
250 }
251
252 iterator end() const noexcept
253 {
254 return iterator { .rest = u8InputString.substr(u8InputString.size()) };
255 }

344/// Converts from UTF-8 to UTF-32. Each ill-formed subsequence becomes one U+FFFD for its maximal subpart, as
345/// Unicode recommends and WHATWG does (E2 82 41 gives U+FFFD A).
346///
347/// @ingroup Unicode
348template <typename T = std::u32string>
349T ToUtf32(std::u8string_view u8InputString)
350{
351 auto result = T {};
352 result.reserve(u8InputString.size());
353 while (!u8InputString.empty())
354 {
355 if (u8InputString.front() < 0x80) // ASCII: one byte, one code point
356 {
357 result.push_back(static_cast<T::value_type>(u8InputString.front()));
358 u8InputString.remove_prefix(1);
359 continue;
360 }
361 auto const [codePoint, length] = detail::Utf32Converter::Decode(u8InputString);
362 result.push_back(static_cast<T::value_type>(codePoint));
363 u8InputString.remove_prefix(length);
364 }
365 return result;
366}
