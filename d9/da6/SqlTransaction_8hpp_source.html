23/// Represents the isolation level of a SQL transaction.
24///
25/// The isolation level determines the degree of isolation of data between concurrent transactions.
26/// The higher the isolation level, the less likely it is that two transactions will interfere with each other.
27///
28/// @see https://learn.microsoft.com/en-us/sql/odbc/reference/develop-app/transaction-isolation-levels
29enum class SqlIsolationMode : std::uint8_t
30{
31 /// The default isolation level of the SQL driver or DBMS
32 DriverDefault = 0,
33
34 /// Transactions are not isolated from each other, allowing to concurrently read uncommitted changes.
35 ReadUncommitted = SQL_TXN_READ_UNCOMMITTED,
36
37 /// The transaction waits until the locked writes are committed by other transactions.
38 ///
39 /// The transaction holds a read lock (if it only reads the row) or write lock (if it updates
40 /// or deletes the row) on the current row to prevent other transactions from updating or deleting it.
41 ReadCommitted = SQL_TXN_READ_COMMITTED,
42
43 /// The transaction waits until the locked writes are committed by other transactions.
44 ///
45 /// The transaction holds read locks on all rows it returns to the application and write locks
46 /// on all rows it inserts, updates, or deletes.
47 RepeatableRead = SQL_TXN_REPEATABLE_READ,
48
49 /// The transaction waits until the locked writes are committed by other transactions.
50 ///
51 /// The transaction holds a read lock (if it only reads rows) or write lock (if it can update
52 /// or delete rows) on the range of rows it affects.
53 Serializable = SQL_TXN_SERIALIZABLE,
54};

78/// @ingroup CoreApi
79/// Represents a transaction to a SQL database.
80///
81/// This class is used to control the transaction manually. It disables the auto-commit mode when constructed,
82/// and automatically commits the transaction when destructed if not done so.
83///
84/// This class is designed with RAII in mind, so that the transaction is automatically committed or rolled back
85/// when the object goes out of scope.
86///
87/// @code
88///
89/// void doSomeWork(SqlTransaction& transaction)
90/// {
91/// auto stmt = SqlStatement { transaction.Connection() };
92/// stmt.ExecuteDirect("INSERT INTO table (column) VALUES (42)"); // Do some work
93/// }
94///
95/// void main()
96/// {
97/// auto connection = SqlConnection {};
98/// auto transaction = SqlTransaction { connection };
99///
100/// doSomeWork(transaction);
101///
102/// transaction.Commit();
103/// }
104/// @endcode
105class SqlTransaction
106{
107 public:
108 SqlTransaction() = delete;
109 SqlTransaction(SqlTransaction const&) = delete;
110 SqlTransaction& operator=(SqlTransaction const&) = delete;

136 /// Try to rollback the transaction, and return true if successful, false otherwise.
137 ///
138 /// A failed rollback leaves the connection in manual-commit mode, because turning autocommit on would
139 /// commit a transaction the driver might still hold; @c SqlConnection::TransactionActive() then
140 /// reports it, and a pool discards such a connection. The exception is a driver that still believes
141 /// in a transaction the database already ended on its own (SQLite after @c INSERT @c OR @c ROLLBACK),
142 /// which the dialect can bring back in line (see
143 /// @c SqlQueryFormatter::ResyncEndedTransactionStatement()): then autocommit is turned back on, and
144 /// false is still returned.
145 LIGHTWEIGHT_API bool TryRollback() noexcept;
146
147 /// Commit the transaction. Throws a SqlTransactionException, naming the driver's diagnostic, if the
148 /// transaction cannot be committed.
149 LIGHTWEIGHT_API void Commit();
150
151 /// Try to commit the transaction, and return true if successful, false otherwise.
152 ///
153 /// A failed commit leaves the transaction to the caller, autocommit still off: a later successful
154 /// TryCommit() ends it, as does a rollback. If the caller does neither, the destructor of a
155 /// @c COMMIT-mode or @c ROLLBACK-mode transaction rolls it back - it does not try the failed commit
156 /// again - while a @c NONE-mode transaction leaves the connection in manual-commit mode, as that
157 /// mode always does.
158 LIGHTWEIGHT_API bool TryCommit() noexcept;
159
160 /// The diagnostic of the last failed commit, rollback or autocommit switch; empty after a success.
161 /// It is read right after the failing call, so it stays available after the driver discarded it.
162 [[nodiscard]] std::optional<SqlErrorInfo> const& LastError() const noexcept
163 {
164 return m_lastError;
165 }

172 }
173
174 /// Ends the transaction as the default mode says: commits (rolling back if that fails), rolls back,
175 /// or, in @c NONE mode, leaves it open.
176 void End() noexcept;
177
178 /// Checks @p result of a transaction call through the connection fault seam, recording the
179 /// diagnostic of a failure unless an earlier one of the same commit or rollback is recorded already.
180 /// @param result The ODBC return code.
181 /// @param location The call site, for the fault seam.
182 /// @return Whether the call succeeded.
183 bool Check(SQLRETURN result, std::source_location location) noexcept;
184
185 /// Turns autocommit back on, recording the driver's diagnostic if that fails.
186 /// @return Whether autocommit is on again.
187 bool RestoreAutoCommit() noexcept;
188
189 /// After a failed rollback: opens a transaction on the database for the driver to end, if the
190 /// dialect knows how (see @c SqlQueryFormatter::ResyncEndedTransactionStatement()), and ends it.
191 /// @return Whether the driver and the database agree again that no transaction is open.
192 bool ResyncEndedTransaction() noexcept;
193
194 private:
195 SqlConnection* m_connection {};
196 SqlTransactionMode m_defaultMode {};
197 std::source_location m_location {};
198 /// The diagnostic of the last failed commit or rollback (see LastError()).
199 std::optional<SqlErrorInfo> m_lastError {};
