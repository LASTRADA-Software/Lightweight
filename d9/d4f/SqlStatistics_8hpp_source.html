5// Optional built-in statistics collection. Part of the Lightweight public API:
6// any consumer (tests, tools, examples, downstream apps) inherits the macros.
7//
8// Configure with `-DLIGHTWEIGHT_ENABLE_STATISTICS=ON`. When the option is OFF,

65/// @ingroup CoreApi
66/// A fixed-bucket latency histogram, in microseconds.
67///
68/// Buckets are power-of-two spaced: bucket `i` counts samples in
69/// `[2^(i-1), 2^i)` microseconds, with bucket 0 counting `[0, 1)` and the last
70/// bucket acting as an open-ended overflow. That covers sub-microsecond calls
71/// up to ~35 minutes in @ref BucketCount buckets, and makes bucket selection a
72/// single `std::bit_width` rather than a search — cheap enough to sit on the
73/// hot path.
74///
75/// This is the plain-struct snapshot type. It is a value: copy it, export it,
76/// diff two of them. The live counters are held separately by @ref SqlStatistics.
77struct SqlLatencyHistogram
78{
79 /// Number of histogram buckets.
80 static constexpr std::size_t BucketCount = 32;
81
82 /// Per-bucket sample counts; bucket `i` covers `[2^(i-1), 2^i)` microseconds.
83 std::array<std::uint64_t, BucketCount> buckets {};
84
85 /// Total number of samples recorded.
86 std::uint64_t count {};
87
88 /// Sum of all sample values, in microseconds.
89 std::uint64_t totalMicroseconds {};
90
91 /// Smallest sample recorded, in microseconds; 0 when @ref count is 0.
92 std::uint64_t minMicroseconds {};
93
94 /// Largest sample recorded, in microseconds; 0 when @ref count is 0.
95 std::uint64_t maxMicroseconds {};
96
97 /// Retrieves the arithmetic mean sample value.
98 ///
99 /// @return The mean, in microseconds; 0.0 when no samples were recorded.
100 [[nodiscard]] LIGHTWEIGHT_API double AverageMicroseconds() const noexcept;
101
102 /// Estimates a percentile from the bucket counts.
103 ///
104 /// The result is the upper bound of the bucket the percentile falls into,
105 /// so it is an over-estimate bounded by a factor of two — the standard
106 /// trade-off for a power-of-two histogram, and accurate enough to spot a
107 /// regression without storing every sample.
108 ///
109 /// @param percentile The percentile to estimate, in `[0.0, 1.0]` (e.g. 0.99 for p99).
110 /// @return The estimated latency, in microseconds; 0 when no samples were recorded.
111 [[nodiscard]] LIGHTWEIGHT_API std::uint64_t PercentileMicroseconds(double percentile) const noexcept;
112
113 /// Retrieves the index of the bucket a given sample value falls into.
114 ///
115 /// @param microseconds The sample value.
116 /// @return The bucket index, clamped to `BucketCount - 1`.
117 [[nodiscard]] static constexpr std::size_t BucketOf(std::uint64_t microseconds) noexcept
118 {
119 // bit_width(0) == 0 -> bucket 0; bit_width(1) == 1 -> bucket 1 ([1,2)); etc.
120 auto width = std::size_t { 0 };
121 for (auto value = microseconds; value != 0; value >>= 1U)
122 ++width;
123 return width < BucketCount ? width : BucketCount - 1;
124 }
125};

129struct SqlOperationStatistics

132 std::uint64_t succeeded {};

134 /// Number of operations that raised an ODBC error.
135 std::uint64_t failed {};
136
137 /// Number of operations that were transparently retried (e.g. a stale prepared statement

154/// @ingroup CoreApi
155/// Aggregated connection-pool counters.
156///
157/// Pool statistics are recorded per @ref SqlStatistics instance. Because `Pool` is a class template
158/// keyed on a compile-time configuration, a process typically holds several *distinct* pool types;
159/// give each the pool-local collector it should report into rather than aggregating them all into
160/// the process-wide instance, unless a combined view is what you want.
161struct SqlPoolStatistics
162{
163 /// Number of connections handed out (from idle, freshly created, or handed off to a waiter).
164 std::uint64_t acquired {};
165
166 /// Of @ref acquired, how many reused an already-open connection rather than creating one.
167 std::uint64_t reused {};
168
169 /// Of @ref acquired, how many had to block or park because the pool was exhausted.
170 std::uint64_t waited {};
171
172 /// Number of connections returned to the pool.
173 std::uint64_t released {};
174
175 /// Number of returned connections destroyed rather than idled, because the pool was over
176 /// capacity (`GrowthStrategy::BoundedOverflow`).
177 std::uint64_t discarded {};
178
179 /// Connections currently sitting idle in the pool.
180 std::uint64_t idle {};
181
182 /// Connections currently checked out of the pool.
183 ///
184 /// @note Only `GrowthStrategy::BoundedWait` tracks a checked-out count — it is what bounds the
185 /// pool. The non-blocking strategies (`UnboundedGrow`, `BoundedOverflow`) never maintain one, so
186 /// this reads 0 for them; use @ref acquired minus @ref released there instead.
187 std::uint64_t checkedOut {};
188
189 /// Distribution of the time spent waiting for a connection to become available. Only
190 /// acquisitions that actually waited contribute a sample.
191 SqlLatencyHistogram waitLatency {};
192
193 /// Retrieves the fraction of acquisitions served by an already-open connection.
194 ///
195 /// @return The reuse rate in `[0.0, 1.0]`; 0.0 when nothing was acquired yet.
196 [[nodiscard]] LIGHTWEIGHT_API double ReuseRate() const noexcept;
197};

199/// @ingroup CoreApi
200/// An immutable, plain-struct view of everything a @ref SqlStatistics collector has observed.
201///
202/// Deliberately free of any exporter-specific concept: read the fields and feed them to Prometheus,
203/// StatsD, a log line, or a test assertion. Obtain one via @ref SqlStatistics::Snapshot.
204///
205/// @note The snapshot is *not* atomic as a whole. Individual counters are read with relaxed
206/// ordering while other threads may still be recording, so two related counters can disagree by a
207/// few samples at the edges. This is intentional: a consistent snapshot would require locking the
208/// hot path. Treat the numbers as monotonically-growing observations, not as a transaction.
209struct SqlStatisticsSnapshot
210{
211 /// Per-operation-class counters, indexed by @ref SqlStatisticsOperation.
212 std::array<SqlOperationStatistics, static_cast<std::size_t>(SqlStatisticsOperation::Count)> operations {};
213
214 /// Connection-pool counters.
215 SqlPoolStatistics pool {};
216
217 /// Number of connections opened.
218 std::uint64_t connectionsOpened {};
219
220 /// Number of connections closed.
221 std::uint64_t connectionsClosed {};
222
223 /// Total number of rows fetched across all statements.
224 std::uint64_t rowsFetched {};
225
226 /// Number of block-prefetch round-trips (one per `SQLFetchScroll` that materialized a row block).
227 std::uint64_t blockFetches {};
228
229 /// Retrieves the counters for one operation class.
230 ///
231 /// @param operation The operation class to read.
232 /// @return A reference to the counters for @p operation.
233 [[nodiscard]] constexpr SqlOperationStatistics const& operator[](SqlStatisticsOperation operation) const noexcept
234 {
235 return operations[static_cast<std::size_t>(operation)];
236 }
237};

380/// @ingroup CoreApi
381/// RAII scope that times its enclosing region and records it as one operation.
382///
383/// Construct at the top of the region; the destructor records the elapsed time. Call @ref Failed to
384/// mark the operation as errored (the destructor still records the latency, so a failing statement
385/// contributes to the distribution rather than silently vanishing from it).
386///
387/// Prefer the @c LIGHTWEIGHT_STATS_SCOPE macro, which compiles the object away entirely when
388/// statistics are disabled.
389class SqlStatisticsScope
390{
391 public:
392 /// Starts timing a region.
393 ///
394 /// @param operation The operation class to record under.
395 explicit SqlStatisticsScope(SqlStatisticsOperation operation) noexcept:
396 _operation { operation },
397 _startedAt { std::chrono::steady_clock::now() },
398 _uncaughtOnEntry { std::uncaught_exceptions() }
399 {
400 }
401
402 SqlStatisticsScope(SqlStatisticsScope const&) = delete;
403 SqlStatisticsScope& operator=(SqlStatisticsScope const&) = delete;
404 SqlStatisticsScope(SqlStatisticsScope&&) = delete;
405 SqlStatisticsScope& operator=(SqlStatisticsScope&&) = delete;
406
407 ~SqlStatisticsScope()
408 {
409 auto const elapsed =
410 std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::steady_clock::now() - _startedAt);
411 // Leaving via an exception *is* the failure signal on this code base: RequireSuccess throws
412 // SqlException on a non-successful ODBC return. Comparing the in-flight exception count
413 // rather than a flag means every throwing path is classified without touching its call site.
414 auto const failed = _failed || std::uncaught_exceptions() > _uncaughtOnEntry;
415 SqlStatistics::Instance().RecordOperation(_operation, elapsed, failed);
416 }
417
418 /// Marks the timed operation as having failed.
419 void Failed() noexcept
420 {
421 _failed = true;
422 }
423
424 /// Records that the timed operation was transparently retried.
425 void Retried() noexcept
426 {
427 SqlStatistics::Instance().RecordRetry(_operation);
428 }
429
430 private:
431 SqlStatisticsOperation _operation;
432 std::chrono::steady_clock::time_point _startedAt;
433 int _uncaughtOnEntry;
434 bool _failed = false;
435};
