18 /// Where the lazy relation loaders of a record obtain the data mapper they run their query on.
19 ///
20 /// Every data mapper carries one and hands it to the loaders it installs, which keep it alive by
21 /// shared ownership. A record may therefore outlive the mapper - or pool lease - it was read
22 /// through, and still load its relations from the same database: a pooled mapper's source borrows
23 /// from its pool, and a plain mapper's source reconnects with its connection string.
24 class RelationLoadSource
25 {
26 public:
27 RelationLoadSource() = default;
28 RelationLoadSource(RelationLoadSource const&) = delete;
29 RelationLoadSource(RelationLoadSource&&) = delete;
30 RelationLoadSource& operator=(RelationLoadSource const&) = delete;
31 RelationLoadSource& operator=(RelationLoadSource&&) = delete;
32 virtual ~RelationLoadSource() = default;
33
34 /// Borrows a data mapper for one load.
35 ///
36 /// Held exclusively until the last copy of the returned pointer is released, which gives the
37 /// mapper back to where it came from (a pool, or this source), so no two loads ever share its
38 /// connection or statement.
39 ///
40 /// @return The borrowed mapper; never null.
41 /// @throws SqlException Connecting failed, when no idle connection was available.
42 [[nodiscard]] virtual std::shared_ptr<DataMapper> Borrow() = 0;
43 };
44
45 /// Wraps @p inner, a source that follows the default connection string (a pool), so that it only
46 /// serves records read while the default was @p connectionString.
47 ///
48 /// @param inner The default-following source to borrow from.
49 /// @param connectionString The connection string the records' mapper was connected with.
50 /// @param generation The @ref SqlConnection::DefaultConnectionStringGeneration that string was
51 /// current under; lets an unchanged default skip the string comparison.
52 /// @return A source whose @c Borrow() throws @ref SqlDefaultConnectionChangedError once the default
53 /// no longer is @p connectionString.
54 [[nodiscard]] LIGHTWEIGHT_API std::shared_ptr<RelationLoadSource> PinToDefaultConnectionString(
55 std::shared_ptr<RelationLoadSource> inner, SqlConnectionString connectionString, std::uint32_t generation);
56
57 /// Makes @p source the relation-load source of @p dataMapper, for the records it reads from now on.
58 ///
59 /// Used by whatever owns a data mapper on behalf of others - a pool - so that records read through
60 /// it load their relations from the same owner.
61 ///
62 /// @param dataMapper The mapper to configure.
63 /// @param source The source its auto-loaders will borrow from; null restores the default choice.
64 inline void AdoptRelationLoadSource(DataMapper& dataMapper, std::shared_ptr<RelationLoadSource> source) noexcept;
