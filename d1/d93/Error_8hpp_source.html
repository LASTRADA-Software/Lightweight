14/// @brief Why a relation could not provide its record(s).
15///
16/// Returned by the relation accessors (`BelongsTo::Record()`, `HasMany::All()`, `HasMany::Count()`, ...)
17/// through @ref RelationResult instead of throwing.
18///
19/// @ingroup DataMapper
20enum class RelationError : std::uint8_t
21{
22 /// No loader is installed and nothing was loaded: the record was built by hand, or read with
23 /// `DataMapperOptions { .loadRelations = false }`.
24 NotConfigured,
25
26 /// There is nothing to load: the foreign key is NULL, or the referenced row does not exist.
27 NotFound,
28
29 /// The default connection string changed since the record was read (see
30 /// @c SqlConnection::SetDefaultConnectionString), so the record belongs to a database the
31 /// application has switched away from. No query is attempted. Re-read the record, or load its
32 /// relations before switching.
33 Outdated,
34
35 /// The load query failed (lost connection, missing table, ...). The driver's diagnostic is reported
36 /// through @c SqlLogger. Unlike the other errors this one is not remembered: the next access retries.
37 QueryFailed,
38};
39
40/// @brief The outcome of a relation access: the requested value, or why it is unavailable.
41///
42/// @ingroup DataMapper
43template <typename T>
44using RelationResult = std::expected<T, RelationError>;
45
46/// @return The name of @p error, e.g. `"Outdated"`.
47constexpr std::string_view to_string(RelationError error) noexcept
48{
49 switch (error)
50 {
51 case RelationError::NotConfigured:
52 return "NotConfigured";
53 case RelationError::NotFound:
54 return "NotFound";
55 case RelationError::Outdated:
56 return "Outdated";
57 case RelationError::QueryFailed:
58 return "QueryFailed";
59 }
60 return "Unknown";
61}

63/// @brief Raised by the relation shortcuts that cannot return a @ref RelationResult - `operator->`,
64/// `operator*`, `begin()`/`end()`, `At()` and `operator[]` - when the relation is unavailable.
65///
66/// Prefer the accessors returning @ref RelationResult (`Record()`, `All()`, `Count()`, ...) to handle
67/// the unavailable case without exceptions.
68///
69/// @ingroup DataMapper
70class SqlRequireLoadedError: public std::runtime_error
71{
72 public:
73 /// Constructs the error with the name of the column type that failed to load.
74 ///
75 /// @param columnType The relation type, for the message.
76 /// @param error Why it failed.
77 explicit SqlRequireLoadedError(std::string_view columnType, RelationError error = RelationError::NotConfigured):
78 std::runtime_error(std::format("Could not load the data record: {} ({})", columnType, to_string(error))),
79 _error { error }
80 {
81 }
82
83 /// @return Why the relation could not be loaded.
84 [[nodiscard]] RelationError Error() const noexcept
85 {
86 return _error;
87 }
88
89 private:
90 RelationError _error;
91};
