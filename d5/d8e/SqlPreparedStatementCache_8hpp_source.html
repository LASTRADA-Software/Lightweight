23/// @ingroup CoreApi
24/// @brief Whether a single @c SqlStatement takes part in its connection's prepared-statement cache.
25///
26/// Statements opt in by default, which only has an effect once the owning connection was given a
27/// non-zero cache capacity (see @c SqlConnection::SetPreparedStatementCacheCapacity). Individual
28/// call sites that must not reuse a plan — for instance a statement that straddles a schema change —
29/// opt out via @c SqlStatement::SetPreparedStatementCaching.
30enum class SqlPreparedStatementCaching : uint8_t
31{
32 /// Reuse a pooled handle when one matches, and hand the handle back to the pool afterwards.
33 Enabled,
34
35 /// Never take a handle from, nor give one to, the connection's cache.
36 Disabled,
37};

39/// @ingroup CoreApi
40/// @brief A bounded LRU pool of already-prepared ODBC statement handles, owned by a @c SqlConnection.
41///
42/// Preparing a statement is a server round-trip on most drivers (MS SQL Server, PostgreSQL). This
43/// cache keeps the @c SQLHSTMT handles of recently prepared queries alive, so re-preparing the same
44/// SQL text on the same connection skips @c SQLPrepare entirely.
45///
46/// A handle is *checked out* while a statement uses it: @ref Acquire removes it from the pool and
47/// @ref Release puts it back. Two statements preparing the same query at the same time therefore each
48/// get their own handle, and both are pooled afterwards (subject to the capacity bound). Eviction is
49/// least-recently-released first, which matters because several backends cap the number of live
50/// prepared statements per session.
51///
52/// @note Not thread-safe, mirroring @c SqlConnection: one connection is used by one thread at a time.
53/// @note A pooled handle holds a query plan derived from the schema as it was at preparation time. A
54/// connection that runs DDL must drop those plans via
55/// @c SqlConnection::ClearPreparedStatementCache — Lightweight's own migration paths do it for you.
56class SqlPreparedStatementCache final
57{
58 public:
59 /// @brief A pooled statement handle together with the parameter count the driver reported for it.
60 struct PreparedHandle
61 {
62 /// The native ODBC statement handle, prepared for the associated query text.
63 SQLHSTMT nativeHandle {};
64
65 /// The number of input parameters @c SQLNumParams reported for that query.
66 SQLSMALLINT parameterCount {};
67 };
68
69 /// @brief Cumulative counters, primarily for tests and diagnostics.
70 struct Statistics
71 {
72 /// Prepare requests served from the pool, i.e. without a @c SQLPrepare round-trip.
73 uint64_t hits {};
74
75 /// Prepare requests that had to issue @c SQLPrepare.
76 uint64_t misses {};
77
78 /// Pooled handles freed because the capacity bound was exceeded.
79 uint64_t evictions {};
80 };

125 /// @brief Takes an idle handle prepared for @p query out of the pool.
126 ///
127 /// The caller owns the returned handle until it hands it back via @ref Release (or frees it).
128 ///
129 /// @param query The exact SQL text the handle must have been prepared with.
130 /// @return The pooled handle, or @c std::nullopt when no idle handle matches.
131 [[nodiscard]] LIGHTWEIGHT_API std::optional<PreparedHandle> Acquire(std::string_view query) noexcept;
132
133 /// @brief Hands a prepared handle back to the pool as the most recently used entry.
134 ///
135 /// The caller must have closed the handle's cursor and unbound its columns beforehand. Ownership
136 /// of @p handle transfers to the cache; when the capacity bound is exceeded — or the cache is
137 /// disabled — the surplus handle is freed right away.
138 ///
139 /// @param query The SQL text @p handle is prepared for.
140 /// @param handle The prepared handle to pool.
141 LIGHTWEIGHT_API void Release(std::string_view query, PreparedHandle handle) noexcept;
142
143 /// Frees every pooled handle, e.g. after DDL invalidated the cached query plans.
144 LIGHTWEIGHT_API void Clear() noexcept;
145
146 private:
147 /// One pooled handle plus the query text it is keyed by. Held in a list so node addresses — and
148 /// therefore the @c string_view keys of @c m_index, which point into @c query — stay stable.
149 struct Entry
150 {
151 std::string query;
152 PreparedHandle handle;
153 };
154
155 using EntryList = std::list<Entry>;
156
157 /// Drops the index entry referring to @p entry (there may be several entries per query text).
158 void EraseFromIndex(EntryList::const_iterator entry) noexcept;
159
160 /// Frees the least recently released handles until at most @c m_capacity remain.
161 void EvictSurplus() noexcept;
162
163 std::size_t m_capacity;
164 EntryList m_entries; // front = most recently used
165 std::unordered_multimap<std::string_view, EntryList::iterator> m_index; // query text -> entry
166 Statistics m_stats {};
167};
